// 本物の骨組み（重みつき）の 3D を、計算で歩かせる試作。
import * as THREE from 'three'
import { GLTFLoader } from 'three/examples/jsm/loaders/GLTFLoader.js'

const BASE = '/.superpowers/preview/avatar/'
const DPR = 2
const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true, preserveDrawingBuffer: true })
const scene = new THREE.Scene()
scene.add(new THREE.HemisphereLight('#ffffff', '#e6dccf', 2.2))
const sun = new THREE.DirectionalLight('#fff2e4', 2.4)
sun.position.set(-0.6, 0.8, 0.55)
scene.add(sun)
const holder = new THREE.Group()
scene.add(holder)
const cam = new THREE.OrthographicCamera()
const ELEV = (35 * Math.PI) / 180

type Rig = { root: THREE.Group; inner: THREE.Object3D; bones: Record<string, THREE.Bone>; rest: Record<string, THREE.Quaternion>; parentRest: Record<string, THREE.Quaternion>; tris: number }
export type Mode = 'idle' | 'walk' | 'run' | 'joy' | 'surprise'
const X = new THREE.Vector3(1, 0, 0)
const Z = new THREE.Vector3(0, 0, 1)
const q1 = new THREE.Quaternion()
const q2 = new THREE.Quaternion()

// 関節の角度（ラジアン）。前に振る＝プラス
export function poseFor(mode: Mode, ph: number) {
  const s = Math.sin(ph * Math.PI * 2)
  const c = Math.cos(ph * Math.PI * 2)
  const p = { thighL: 0, thighR: 0, kneeL: 0, kneeR: 0, armL: 0, armR: 0, elbow: 0.15, armOut: 0, y: 0, lean: 0, sway: 0, head: 0 }
  if (mode === 'walk' || mode === 'run') {
    const run = mode === 'run'
    const a = run ? 0.85 : 0.5
    p.thighL = a * s + (run ? 0.15 : 0)
    p.thighR = -a * s + (run ? 0.15 : 0)
    p.kneeL = (run ? 1.3 : 0.7) * Math.max(0, c) + (run ? 0.25 : 0.05)
    p.kneeR = (run ? 1.3 : 0.7) * Math.max(0, -c) + (run ? 0.25 : 0.05)
    p.armL = -a * 0.9 * s
    p.armR = a * 0.9 * s
    p.elbow = run ? 1.2 : 0.35
    p.y = (run ? 0.05 : 0.018) * Math.abs(c)
    p.lean = run ? 0.2 : 0.04
    p.sway = 0.04 * s
    p.head = run ? -0.12 : 0
  } else if (mode === 'joy') {
    p.armOut = 2.3
    p.y = 0.15 * Math.abs(s)
    p.head = -0.1
  } else if (mode === 'surprise') {
    p.armOut = 0.5
    p.lean = -0.08
  }
  return p
}

function setBone(r: Rig, name: string, ...rots: [THREE.Vector3, number][]) {
  const b = r.bones[name]
  if (!b) return
  q1.identity()
  for (const [axis, ang] of rots) q1.multiply(q2.setFromAxisAngle(axis, ang))
  // 親の「もとの向き」の中で、世界の軸まわりに回す
  b.quaternion.copy(r.parentRest[name]!).invert().multiply(q1).multiply(r.rest[name]!)
}

export function applyPose(r: Rig, p: ReturnType<typeof poseFor>) {
  setBone(r, 'thighL', [X, -p.thighL])
  setBone(r, 'thighR', [X, -p.thighR])
  setBone(r, 'shinL', [X, p.kneeL])
  setBone(r, 'shinR', [X, p.kneeR])
  setBone(r, 'upperarmL', [Z, p.armOut], [X, -p.armL])
  setBone(r, 'upperarmR', [Z, -p.armOut], [X, -p.armR])
  setBone(r, 'forearmL', [X, -p.elbow])
  setBone(r, 'forearmR', [X, -p.elbow])
  setBone(r, 'chest', [X, p.lean], [Z, p.sway])
  setBone(r, 'head', [X, p.head - p.lean * 0.6], [Z, -p.sway])
  r.inner.position.y = p.y
}

async function load(url: string): Promise<Rig> {
  const gltf = await new GLTFLoader().loadAsync(url)
  const root = new THREE.Group()
  root.add(gltf.scene)
  gltf.scene.updateMatrixWorld(true)
  const bones: Rig['bones'] = {}
  const rest: Rig['rest'] = {}
  const parentRest: Rig['parentRest'] = {}
  let tris = 0
  gltf.scene.traverse((o) => {
    const m = o as THREE.SkinnedMesh
    if (m.isMesh) {
      m.material = new THREE.MeshStandardMaterial({ vertexColors: !!m.geometry.attributes.color, map: (m.material as THREE.MeshStandardMaterial).map ?? null, roughness: 0.9, metalness: 0 })
      m.frustumCulled = false
      tris += (m.geometry.index ? m.geometry.index.count : m.geometry.attributes.position!.count) / 3
    }
    if ((o as THREE.Bone).isBone) {
      bones[o.name] = o as THREE.Bone
      rest[o.name] = o.getWorldQuaternion(new THREE.Quaternion())
      parentRest[o.name] = o.parent!.getWorldQuaternion(new THREE.Quaternion())
    }
  })
  const sh = new THREE.Mesh(new THREE.CircleGeometry(0.22, 24), new THREE.MeshBasicMaterial({ color: '#82481f', transparent: true, opacity: 0.2, depthWrite: false }))
  sh.rotation.x = -Math.PI / 2
  sh.position.y = 0.003
  root.add(sh)
  return { root, inner: gltf.scene, bones, rest, parentRest, tris }
}

function shot(r: Rig, w: number, h: number, heightPx: number, yawDeg: number, elev = ELEV, bg = '#fbf1d6') {
  holder.clear()
  holder.add(r.root)
  r.root.rotation.y = (yawDeg * Math.PI) / 180
  const up = new THREE.Vector3(0, Math.cos(elev), -Math.sin(elev))
  const upp = Math.cos(elev) / heightPx
  cam.left = (-w / 2) * upp
  cam.right = (w / 2) * upp
  cam.top = (h / 2) * upp
  cam.bottom = (-h / 2) * upp
  cam.near = -10
  cam.far = 10
  const target = up.clone().multiplyScalar((h / 2 - h * 0.13) * upp)
  cam.position.copy(target).add(new THREE.Vector3(0, Math.sin(elev), Math.cos(elev)))
  cam.up.set(0, 1, 0)
  cam.lookAt(target)
  cam.updateProjectionMatrix()
  renderer.setPixelRatio(1)
  renderer.setSize(w * DPR, h * DPR, false)
  renderer.setClearColor(bg, 1)
  renderer.render(scene, cam)
  const cv = document.createElement('canvas')
  cv.width = w * DPR
  cv.height = h * DPR
  cv.style.width = `${w}px`
  cv.style.height = `${h}px`
  cv.getContext('2d')!.drawImage(renderer.domElement, 0, 0)
  return cv
}

const app = document.getElementById('app')!
function sheet(id: string, title: string, note: string) {
  const d = document.createElement('div')
  d.className = 'sheet'
  d.id = id
  d.innerHTML = `<h2>${title}</h2><p class="note">${note}</p>`
  app.append(d, document.createElement('br'))
  return d
}
function row(parent: HTMLElement, label: string) {
  const r = document.createElement('div')
  r.className = 'row'
  r.innerHTML = `<div class="lab">${label}</div>`
  parent.append(r)
  return r
}
function tile(parent: HTMLElement, cv: HTMLCanvasElement, label = '') {
  const t = document.createElement('div')
  t.className = 'tile'
  t.append(cv, label)
  parent.append(t)
}

async function main() {
  await document.fonts.load('12px MP')
  const file = new URLSearchParams(location.search).get('glb') ?? 'ai-rigged.glb'
  const r = await load(BASE + file)
  const s = sheet('rig', '骨組みを入れた 3D を計算で歩かせる', `${file}（三角形 ${r.tris}）。ひざ・ひじも曲がる`)
  const PH = [0, 0.125, 0.25, 0.375, 0.5, 0.625, 0.75, 0.875]
  for (const [label, mode, yaw, elev] of [['歩き（真横）', 'walk', 90, 0.12], ['走り（真横）', 'run', 90, 0.12], ['歩き（地図の見え方・ななめ前）', 'walk', 35, ELEV], ['走り（地図の見え方・ななめ後ろ）', 'run', 140, ELEV]] as const) {
    const rr = row(s, label)
    for (const ph of PH) {
      applyPose(r, poseFor(mode, ph))
      tile(rr, shot(r, 150, 215, 165, yaw, elev))
    }
  }
  let rr = row(s, 'ポーズ')
  for (const [label, mode, ph] of [['立ち', 'idle', 0], ['ばんざい', 'joy', 0.25], ['びっくり', 'surprise', 0]] as const) {
    applyPose(r, poseFor(mode, ph))
    tile(rr, shot(r, 190, 270, 190, 0, 0.12), label)
  }
  applyPose(r, poseFor('idle', 0))
  for (const yaw of [0, 90, 180]) tile(rr, shot(r, 190, 270, 190, yaw, 0.12), `立ち ${yaw}°`)
  rr = row(s, '実際の大きさ 40px・歩き')
  for (const ph of PH) {
    applyPose(r, poseFor('walk', ph))
    tile(rr, shot(r, 150, 62, 40, 35))
  }
  // 重さ
  const t0 = performance.now()
  for (let i = 0; i < 20000; i++) applyPose(r, poseFor('walk', i / 30))
  ;(window as unknown as { __stats: unknown }).__stats = { tris: r.tris, bones: Object.keys(r.bones).length, poseMicroSec: Number((((performance.now() - t0) / 20000) * 1000).toFixed(2)) }
  document.body.dataset.ready = '1'
}
main().catch((e) => {
  document.body.append(String(e))
  console.error(e)
})
